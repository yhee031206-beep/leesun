  <script>
    function updateClock() {
      const now = new Date();
      const hours = String(now.getHours()).padStart(2, '0');
      const minutes = String(now.getMinutes()).padStart(2, '0');
      const timeString = `${hours}:${minutes}`;
      
      document.getElementById('status-time').textContent = timeString;
      document.getElementById('widget-time').textContent = timeString;

      const days = ['일', '월', '화', '수', '목', '금', '토'];
      const month = now.getMonth() + 1;
      const date = now.getDate();
      const dayName = days[now.getDay()];
      
      document.getElementById('widget-date').textContent = `${month}월 ${date}일 (${dayName})`;
    }

    function openApp(appName) {
      const modal = document.getElementById('appModal');
      document.getElementById('modalTitle').textContent = appName;
      document.getElementById('modalBody').textContent = `${appName} 앱이 실행되었습니다. 이 공간에 원하시는 스토리나 데이터를 추가하실 수 있습니다.`;
      modal.classList.add('active');
    }

    function closeApp() {
      document.getElementById('appModal').classList.remove('active');
    }

    setInterval(updateClock, 1000);
    updateClock();
  </script>
